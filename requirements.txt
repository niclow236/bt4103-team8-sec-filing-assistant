# Core dependencies for the SEC filing assistant.
# Install with:  pip install -r requirements.txt
# Versions are pinned so everyone on the team runs the same code.

# SEC EDGAR access: company lookup, filing download, and 10-K/10-Q section parsing.
edgartools==5.56.0

# Reads the local .env file so API keys and the EDGAR identity stay out of the repo.
python-dotenv==1.2.3

# Sparse keyword retrieval baseline.
rank-bm25==0.2.2

# Sentence embeddings for the dense retriever, and the store the vectors go in.
# sentence-transformers pulls in torch, which is most of the install size. The
# CPU build is enough, since nobody on the team is training anything, but it is
# not fast: the first full build of the dense index took 6.6 hours on a laptop
# CPU, before a 1.19x speed-up from sorting by length. embed is incremental, so
# that cost is paid once rather than per re-chunk.
sentence-transformers==6.0.1
chromadb==1.5.9

# The embedding model's tokenizer on its own, without torch, so verify can
# count every passage in the model's tokens in seconds. Already installed as a
# dependency of sentence-transformers; pinned here because verify uses it
# directly.
tokenizers==0.23.2

# Tests: python -m pytest
pytest==9.1.1
