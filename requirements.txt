# Core dependencies for the SEC filing assistant.
# Install with:  pip install -r requirements.txt
# Versions are pinned so everyone on the team runs the same code.

# SEC EDGAR access: company lookup, filing download, and 10-K/10-Q section parsing.
edgartools==5.56.0

# Reads the local .env file so the EDGAR identity and machine settings stay out of the repo.
python-dotenv==1.2.3

# Sparse keyword retrieval baseline.
rank-bm25==0.2.2

# Sentence embeddings for the dense retriever, and the store the vectors go in.
# sentence-transformers pulls in torch, which is most of the install size. The
# CPU build is enough, since nobody on the team is training anything, but it is
# not fast: the first full build of the dense index took 6.6 hours on a laptop
# CPU, before a 1.19x speed-up from sorting by length. embed is incremental, so
# that cost is paid once rather than per re-chunk.
sentence-transformers==6.0.1
chromadb==1.5.9

# What the embedding and reranking models run on, and the arrays the scores come
# back in. Both are installed as dependencies of sentence-transformers, which
# accepts a range of each; pinned here at the versions the reported results were
# measured with (docs/evaluation/full-matrix-20261021-manifest.json records
# both), since embed, rerank and the C5 run use them directly (#50). numpy 2.3 and later need Python 3.11, so
# 3.10 gets the last release that supports it. CI installs torch from PyTorch's
# CPU index before this file, reading its version from the line below.
torch==2.14.0
numpy==2.5.3; python_version >= "3.11"
numpy==2.2.6; python_version < "3.11"

# The embedding model's tokenizer on its own, without torch, so verify can
# count every passage in the model's tokens in seconds. Already installed as a
# dependency of sentence-transformers; pinned here because verify uses it
# directly.
tokenizers==0.23.2

# Process-safe experiment locking; already installed through torch.
filelock==3.32.6

# What sentence-transformers loads the embedding model with. Already installed
# as its dependency, which accepts any 5.x; pinned here because the app works
# around how this version registers its alias modules (_settle_lazy_aliases
# in src/app/main.py). Moving this pin, or streamlit's below, is when to look
# at that again, and tests/test_app_live.py says which way it went.
transformers==5.17.0

# Answer generation, by a local model served by Ollama (the default), by
# Mistral's API on its free plan, with each member's own key, or by NUS SoC's
# SOCLAAS API with a key from its portal (see the README).
# Ollama itself is not a Python package: install it from
# https://ollama.com/download and pull a model.
#
# LangChain's chat models are the clients, streaming and token counts included:
# langchain-ollama for Ollama, langchain-mistralai for Mistral and
# langchain-openai for SOCLAAS, which speaks OpenAI's API. langchain-core is
# what generate reads partial JSON with; langchain-openai 1.7 needs 1.6.10 or
# later. ollama and openai are the client libraries underneath, pinned because
# generate catches their errors. openai sends with httpx2, not httpx, and
# generate reads the httpx2 error a SOCLAAS failure wraps.
langchain-core==1.6.12
langchain-ollama==1.1.0
langchain-mistralai==1.1.6
langchain-openai==1.7.2
ollama==0.6.2
openai==3.24.0
httpx2==2.13.0

# The schema an answer is generated against and validated with. Already
# installed as a dependency of chromadb and langchain-core; pinned here because
# rag/records.py uses it directly.
pydantic==2.13.5

# HTTP client underneath both chat models. Already installed as a dependency of
# edgartools and chromadb; pinned here because generate catches its errors and
# the tests mock Ollama's server and Mistral's API with it.
httpx==0.28.1
# How langchain-mistralai reads Mistral's event stream. Installed as its
# dependency; pinned here because generate catches its SSEError.
httpx-sse==0.4.3

# The Mistral test notebook (notebooks/mistral/), which also uses
# langchain-mistralai above:
# - ipykernel: runs the notebook in VS Code or Jupyter.
# - nbconvert: runs it from a terminal or an agent (jupyter nbconvert --execute).
# - openpyxl: refreshes the test questions from the team's workbook.
ipykernel==7.3.0
nbconvert==7.17.1
openpyxl==3.1.5

# Tests: python -m pytest
pytest==9.1.1

# Interactive answer cards, citation expanders and metadata filters.
streamlit==1.64.0
# The Results page's chart is built as an Altair chart rather than left to
# st.bar_chart, so its bars can be coloured and labelled one at a time (#166).
# Streamlit installs altair anyway; pinned because components.py imports it.
altair==6.2.2

# Tables: the facts store is a parquet file, and the app, the benchmark and the
# answer checks read it as a DataFrame. edgartools and streamlit install both
# anyway; pinned because the code imports pandas, and the tests build a table
# with pyarrow (#50). pandas 3 needs Python 3.11, so 3.10 gets the last 2.x.
pandas==3.0.5; python_version >= "3.11"
pandas==2.3.3; python_version < "3.11"
pyarrow==25.0.1

# Tests read the HTML the app renders with BeautifulSoup. edgartools installs
# it anyway; pinned because the tests import it (#50).
beautifulsoup4==4.15.0

# Test coverage: python -m pytest --cov, with the settings in .coveragerc.
# pytest-cov runs coverage inside pytest. coverage is pinned as well, because
# .coveragerc is its configuration and everyone's report should come from the
# same version.
pytest-cov==7.1.0
coverage==7.16.2
